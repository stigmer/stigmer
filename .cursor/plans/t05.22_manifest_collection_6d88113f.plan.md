- Prevents code duplication
- Maintains existing architecture consistency
- Leverages the already-working Synthesize() flow

The `synthesis.Result` struct is the de facto `ManifestSet` - we just need to extend it.

import mcpserverv1 "github.com/stigmer/stigmer/apis/stubs/go/ai/stigmer/agentic/mcpserver/v1"

    // from ApiResourceReference fields via proto reflection. See Phase 5 architecture docs.
    Dependencies map[string][]string

func (r *Result) McpServerCount() int {
    return len(r.McpServers)
}

func GetResourceID(msg proto.Message) string {
    switch m := msg.(type) {
    // ... existing cases ...
    case *mcpserverv1.McpServer:
        slug := m.GetMetadata().GetSlug()
        if slug == "" {
