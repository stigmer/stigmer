## What you'll build

A working Agent you can call from your code — in 5 minutes. You will connect to
Stigmer Cloud, send a message, and get a response through the SDK.

**Time**: ~5 minutes **Prerequisites**: A browser and one of:
[Node.js](https://nodejs.org/) 20+, [Go](https://go.dev/) 1.21+,
[Python](https://python.org/) 3.11+, or [Java](https://jdk.java.net/) 17+

<Callout type="info">
  Want to run locally without a cloud account? See the [Local
  Quickstart](/docs/getting-started/local).
</Callout>

## Sign up and get your API key

1. Go to [`app.stigmer.ai`](https://app.stigmer.ai) and create an account.
2. Open **Settings → API Keys** and click **+ New API key**.
3. Copy the key.

<Callout type="info">
  Your API key starts with `sk_`. Keep it secret. Never commit it to source
  control.

## Create a Session and send a message

A Session is a conversation with your Agent. Create one, then send a message to
start a run. You don't specify which Agent to use — Stigmer's built-in assistant
handles the request automatically.

Add the following to the same file you created in the previous step.

## Stream the response

Follow the Session's events to receive the Agent's response as it arrives. A
Session's output is one ordered log of events. Streaming it from the start
(`after_seq` 0) sends every event already stored, then each new one as the Agent
works. Print the Agent's messages for your run, and stop when the run's
`stigmer.run.status` event says it ended.

Add the following to the same file, right after the code from the previous step.

<Tabs items={["TypeScript", "Go", "Python", "Java"]} groupId="sdk-language" persist>
<Tab value="TypeScript">

Add to `ask-agent.ts`:

```typescript
  const ended = ["completed", "failed", "cancelled", "terminated"];
  for await (const frame of followSession(stigmer, session.metadata!.id)) {
    if (frame.frame.case !== "event") continue;
    const { runId, threadId, event } = frame.frame.value;
    if (runId !== run.metadata!.id) continue;
    if (event.case === "agentMessage" && threadId === "") {
      console.log(event.value.content.map((block) => block.text).join(""));
    }
    if (event.case === "stigmerRunStatus" && ended.includes(event.value.phase)) {
      break;
    }
  }
}

main();
```

</Tab>
<Tab value="Go">

Add to `main.go`:

```go
	stream, err := client.Session.StreamEvents(ctx, &sessionv1.StreamSessionEventsRequest{
		SessionId: session.GetMetadata().GetId(),
		AfterSeq:  proto.Int64(0),
	})
	if err != nil {
		log.Fatal(err)
	}

	for {
		frame, err := stream.Recv()
		if err == io.EOF {
			break
		}
		if err != nil {
			log.Fatal(err)
		}
		event := frame.GetEvent()
		if event.GetRunId() != run.GetMetadata().GetId() {
			continue
		}
		if msg := event.GetAgentMessage(); msg != nil && event.GetThreadId() == "" {
			for _, block := range msg.GetContent() {
				fmt.Print(block.GetText())
			}
			fmt.Println()
		}
		if status := event.GetStigmerRunStatus(); status != nil {
			switch status.GetPhase() {
			case "completed", "failed", "cancelled", "terminated":
				return
			}
		}
	}
}
```

package main

import (
	"context"
	"fmt"
	"io"
	"log"
	"os"
	"time"

	stigmer "github.com/stigmer/stigmer/sdk/go/v3"
	sessionv1 "github.com/stigmer/stigmer/sdk/go/v3/proto/ai/stigmer/agentic/session/v1"
	"google.golang.org/protobuf/proto"
)

func main() {
	client, err := stigmer.NewClient(stigmer.WithAPIKey(os.Getenv("STIGMER_API_KEY")))
	if err != nil {
		log.Fatal(err)
	}
	defer client.Close()

	ctx := context.Background()
	ts := fmt.Sprintf("%d", time.Now().UnixMilli())

	session, err := client.Session.Create(ctx, &stigmer.SessionInput{
		Name: "session-" + ts,
		Org:  "my-org",
	})
	if err != nil {
		log.Fatal(err)
	}

	run, err := client.Run.Create(ctx, &stigmer.RunInput{
		Name:      "run-" + ts,
		Org:       "my-org",
		SessionId: session.GetMetadata().GetId(),
		Message:   "What are some best practices for handling customer complaints?",
	})
	if err != nil {
		log.Fatal(err)
	}

	stream, err := client.Session.StreamEvents(ctx, &sessionv1.StreamSessionEventsRequest{
		SessionId: session.GetMetadata().GetId(),
		AfterSeq:  proto.Int64(0),
	})
	if err != nil {
		log.Fatal(err)
	}

	for {
		frame, err := stream.Recv()
		if err == io.EOF {
			break
		}
		if err != nil {
			log.Fatal(err)
		}
		event := frame.GetEvent()
		if event.GetRunId() != run.GetMetadata().GetId() {
			continue
		}
		if msg := event.GetAgentMessage(); msg != nil && event.GetThreadId() == "" {
			for _, block := range msg.GetContent() {
				fmt.Print(block.GetText())
			}
			fmt.Println()
		}

			switch status.GetPhase() {
			case "completed", "failed", "cancelled", "terminated":

			}
		}
	}

import ai.stigmer.agentic.session.v1.StreamSessionEventsRequest;
import ai.stigmer.sdk.StigmerClient;
import ai.stigmer.sdk.gen.SessionInput;
import ai.stigmer.sdk.gen.RunInput;
import java.util.Set;

public class AskAgent {
    public static void main(String[] args) {
        var client = StigmerClient.builder(
            System.getenv("STIGMER_API_KEY")
        ).build();

        var ts = String.valueOf(System.currentTimeMillis());

        var session = client.session.create(
            SessionInput.builder()
                .name("session-" + ts)
                .org("my-org")
                .build()
        );

        var run = client.run.create(
            RunInput.builder()
                .name("run-" + ts)
                .org("my-org")
                .sessionId(session.getMetadata().getId())
                .message("What are some best practices for handling customer complaints?")
                .build()
        );

        var request = StreamSessionEventsRequest.newBuilder()
            .setSessionId(session.getMetadata().getId())
            .setAfterSeq(0)
            .build();
        var ended = Set.of("completed", "failed", "cancelled", "terminated");

        var stream = client.session.streamEvents(request);
        while (stream.hasNext()) {
            var event = stream.next().getEvent();
            if (!event.getRunId().equals(run.getMetadata().getId())) {
                continue;
            }
            if (event.hasAgentMessage() && event.getThreadId().isEmpty()) {
                for (var block : event.getAgentMessage().getContentList()) {
                    System.out.print(block.getText());
                }
                System.out.println();
            }
            if (event.hasStigmerRunStatus()
                    && ended.contains(event.getStigmerRunStatus().getPhase())) {
                break;
            }
        }
    }
}

## What just happened

You called an Agent from your own code through a real API. No Agent
configuration. No infrastructure setup. The built-in assistant handled your
request automatically.

The Agent is good at general knowledge — ask about best practices for handling
complaints, and it gives a solid answer. But change the question to "What is
your return policy for defective items?" and it can only guess. It doesn't know
your return policy, your product catalog, or your escalation process.

In the next guide, you will teach your Agent domain knowledge through the
Stigmer web app. Same question, completely different answer.
