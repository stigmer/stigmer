---
title: Streaming
description:
  Consume real-time Agent Run and Workflow Run updates as they arrive.
---

The `subscribe` method opens a server stream that pushes run snapshots as the
agent works. Each snapshot is the **complete** run state at that point in time —
not a diff or a delta. The stream closes when the run reaches a terminal phase.

If you are new to streaming, the
[Quickstart](/docs/getting-started/quickstart) walks through a minimal subscribe
loop. This guide covers the production patterns you need after that.

## Subscribe to an Agent Run

Call `subscribe` with the run ID after creating a run. Iterate the stream and
check `status.phase` on each snapshot to decide what to do.

<SDKTabs>
<Tab value="TypeScript">
```typescript
for await (const snapshot of stigmer.agentRun.subscribe(runId)) {
  const phase = snapshot.status?.phase;
  const messages = snapshot.status?.messages ?? [];

  if (phase === "RUN_COMPLETED" || phase === "RUN_FAILED") {
    const last = messages[messages.length - 1];
    if (last?.content) {
      console.log(last.content);
    }
    if (phase === "RUN_FAILED") {
      console.error("Run failed:", snapshot.status?.error);
    }
    break;
  }
}
```
</Tab>
<Tab value="Go">
```go
stream, err := client.AgentRun.Subscribe(ctx, runID)
if err != nil {
    log.Fatal(err)
}

for {
    snapshot, err := stream.Recv()
    if err == io.EOF {
        break
    }
    if err != nil {
        log.Fatal(err)
    }

    status := snapshot.GetStatus()
    phase := status.GetPhase()

    if phase == agentrunv1.RunPhase_RUN_COMPLETED || phase == agentrunv1.RunPhase_RUN_FAILED {
        msgs := status.GetMessages()
        if len(msgs) > 0 {
            fmt.Println(msgs[len(msgs)-1].GetContent())
        }
        if phase == agentrunv1.RunPhase_RUN_FAILED {
            log.Fatalf("Run failed: %s", status.GetError())
        }
        break
    }
}
```
</Tab>
<Tab value="Python">
```python
for snapshot in client.agent_runs.subscribe(run_id):
    phase = snapshot.status.phase
    messages = snapshot.status.messages

    if phase in ("RUN_COMPLETED", "RUN_FAILED"):
        if messages and messages[-1].content:
            print(messages[-1].content)
        if phase == "RUN_FAILED":
            print(f"Run failed: {snapshot.status.error}")
        break
```
</Tab>
<Tab value="Java">
```java
var stream = client.agentRun.subscribe(runId);

while (stream.hasNext()) {
    var snapshot = stream.next();
    var status = snapshot.getStatus();
    var phase = status.getPhase();

    if (phase == RunPhase.RUN_COMPLETED || phase == RunPhase.RUN_FAILED) {
        var messages = status.getMessagesList();
        if (!messages.isEmpty()) {
            System.out.println(messages.get(messages.size() - 1).getContent());
        }
        if (phase == RunPhase.RUN_FAILED) {
            System.err.println("Run failed: " + status.getError());
        }
        break;
    }
}
```
</Tab>
</SDKTabs>

## Read the snapshot

Every snapshot is a full `AgentRun` resource. The fields you inspect most
often are on `status`:

| Field | What it contains |
|-------|------------------|
| `phase` | Current lifecycle phase. See [Detect completion](#detect-completion). |
| `messages` | Conversation history — human messages, AI responses, tool results, and system notices. Each message has a `type`, `content`, and optional `toolCalls`. |
| `error` | Failure reason. Populated when phase is `RUN_FAILED` or `RUN_TERMINATED`. |
| `pendingApprovals` | Tool calls waiting for human approval. See [Connect Tools](/docs/getting-started/connect-tools) for the full approval handling pattern. |
| `todos` | Task list the agent maintains to track multi-step work. |
| `subAgentRuns` | Child agent invocations with their own messages and status. |
| `artifacts` | Files the agent published for download. |
| `setupProgress` | Setup phase label, reported while phase is `RUN_PENDING`. |

<Callout type="info">
  For the complete field reference, see
  [Agent Run > AgentRunStatus](/docs/sdk/resources/agent-run#agentrunstatus).
</Callout>

## Detect completion

Run phases fall into two categories:

**Terminal** — the run is finished and the stream will close:

| Phase | Meaning |
|-------|---------|
| `RUN_COMPLETED` | Agent finished successfully. |
| `RUN_FAILED` | Agent encountered an error. |
| `RUN_CANCELLED` | User stopped the run gracefully. |
| `RUN_TERMINATED` | Platform stopped the run (timeout, budget, or force-kill). |

**Non-terminal** — the run is still active or paused:

| Phase | Meaning |
|-------|---------|
| `RUN_PENDING` | Queued, waiting for a worker. |
| `RUN_IN_PROGRESS` | Agent is working. |
| `RUN_WAITING_FOR_APPROVAL` | Blocked on a human approval decision. |
| `RUN_PAUSED` | Paused by the user, can be resumed. |

Check for any terminal phase to decide when to stop processing:

<SDKTabs>
<Tab value="TypeScript">
```typescript
const TERMINAL_PHASES = new Set([
  "RUN_COMPLETED",
  "RUN_FAILED",
  "RUN_CANCELLED",
  "RUN_TERMINATED",
]);

function isTerminal(phase: string | undefined): boolean {
  return TERMINAL_PHASES.has(phase ?? "");
}
```
</Tab>
<Tab value="Go">
```go
func isTerminal(phase agentrunv1.RunPhase) bool {
    switch phase {
    case agentrunv1.RunPhase_RUN_COMPLETED,
        agentrunv1.RunPhase_RUN_FAILED,
        agentrunv1.RunPhase_RUN_CANCELLED,
        agentrunv1.RunPhase_RUN_TERMINATED:
        return true
    }
    return false
}
```
</Tab>
<Tab value="Python">
```python
TERMINAL_PHASES = {
    "RUN_COMPLETED",
    "RUN_FAILED",
    "RUN_CANCELLED",
    "RUN_TERMINATED",
}

def is_terminal(phase: str) -> bool:
    return phase in TERMINAL_PHASES
```
</Tab>
<Tab value="Java">
```java
static boolean isTerminal(RunPhase phase) {
    return phase == RunPhase.RUN_COMPLETED
        || phase == RunPhase.RUN_FAILED
        || phase == RunPhase.RUN_CANCELLED
        || phase == RunPhase.RUN_TERMINATED;
}
```
</Tab>
</SDKTabs>

## Handle stream errors

Transport errors (network failures, server unavailability) surface as SDK
exceptions during iteration. Wrap the subscribe loop to catch them separately
from run-level failures reported in `status.error`.

<SDKTabs>
<Tab value="TypeScript">
```typescript
import { StigmerError, isRetryable } from "@stigmer/sdk";

try {
  for await (const snapshot of stigmer.agentRun.subscribe(runId)) {
    // process snapshot
  }
} catch (err) {
  if (isRetryable(err)) {
    // transient failure — wait and re-subscribe
  }
  throw err;
}
```
</Tab>
<Tab value="Go">
```go
stream, err := client.AgentRun.Subscribe(ctx, runID)
if err != nil {
    log.Fatal(err)
}

for {
    snapshot, err := stream.Recv()
    if err == io.EOF {
        break // stream closed normally
    }
    if err != nil {
        if stigmer.IsRetryable(err) {
            // transient failure — wait and re-subscribe
            break
        }
        log.Fatal(err)
    }
    // process snapshot
}
```
</Tab>
<Tab value="Python">
```python
from stigmer import StigmerError, is_retryable

try:
    for snapshot in client.agent_runs.subscribe(run_id):
        # process snapshot
        pass
except StigmerError as err:
    if is_retryable(err):
        pass  # transient failure — wait and re-subscribe
    raise
```
</Tab>
<Tab value="Java">
```java
try {
    var stream = client.agentRun.subscribe(runId);
    while (stream.hasNext()) {
        var snapshot = stream.next();
        // process snapshot
    }
} catch (StigmerException e) {
    if (e.isRetryable()) {
        // transient failure — wait and re-subscribe
    }
    throw e;
}
```
</Tab>
</SDKTabs>

Run failures are not transport errors. When an agent fails, the stream
delivers a snapshot with `phase` set to `RUN_FAILED` and `error`
containing the failure reason. The stream then closes normally.

For more on error classification, see
[SDK Overview > Error handling](/docs/sdk/overview#error-handling).

## Cancel a stream

To stop consuming updates before the run finishes, cancel the stream from
the client side.

<SDKTabs>
<Tab value="TypeScript">

Pass an `AbortSignal` to `subscribe`. Abort the controller to close the stream:

```typescript
const controller = new AbortController();
setTimeout(() => controller.abort(), 30_000);

for await (const snapshot of stigmer.agentRun.subscribe(
  runId,
  controller.signal,
)) {
  // process snapshot
}
```

</Tab>
<Tab value="Go">

Cancel the `context.Context` to close the stream:

```go
ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
defer cancel()

stream, err := client.AgentRun.Subscribe(ctx, runID)
if err != nil {
    log.Fatal(err)
}

for {
    snapshot, err := stream.Recv()
    if err != nil {
        break
    }
    // process snapshot
}
```

</Tab>
<Tab value="Python">

Break out of the iterator to stop reading. The underlying gRPC channel cleans
up when the iterator is garbage collected:

```python
for snapshot in client.agent_runs.subscribe(run_id):
    if should_stop():
        break
```

</Tab>
<Tab value="Java">

Break out of the iteration loop. The underlying gRPC stream cleans up when the
`StigmerStream` is garbage collected:

```java
var stream = client.agentRun.subscribe(runId);
while (stream.hasNext()) {
    var snapshot = stream.next();
    if (shouldStop()) {
        break;
    }
}
```

</Tab>
</SDKTabs>

<Callout type="warn">
  Cancelling a stream stops your client from receiving updates. It does not
  cancel the run itself. To cancel the run, call
  [`cancel`](/docs/sdk/resources/agent-run#cancel) or
  [`terminate`](/docs/sdk/resources/agent-run#terminate).
</Callout>

## Subscribe to a Workflow Run

Workflow Run streams work the same way, with two differences:

1. `subscribe` takes a request object with a `runId` field instead of a
   bare string.
2. Progress is tracked through `status.tasks` (a list of workflow tasks with
   individual status) rather than `status.messages`.

<SDKTabs>
<Tab value="TypeScript">
```typescript
for await (const snapshot of stigmer.workflowRun.subscribe(
  { runId: workflowRunId },
)) {
  const phase = snapshot.status?.phase;
  const tasks = snapshot.status?.tasks ?? [];

  for (const task of tasks) {
    console.log(`${task.taskName}: ${task.status}`);
  }

  if (isTerminal(phase)) {
    if (phase === "RUN_COMPLETED") {
      console.log("Output:", snapshot.status?.output);
    }
    break;
  }
}
```
</Tab>
<Tab value="Go">
```go
stream, err := client.WorkflowRun.Subscribe(ctx,
    &workflowrunv1.SubscribeWorkflowRunRequest{
        RunId: workflowRunID,
    })
if err != nil {
    log.Fatal(err)
}

for {
    snapshot, err := stream.Recv()
    if err == io.EOF {
        break
    }
    if err != nil {
        log.Fatal(err)
    }

    status := snapshot.GetStatus()
    for _, task := range status.GetTasks() {
        fmt.Printf("%s: %s\n", task.GetTaskName(), task.GetStatus())
    }

    if isTerminal(status.GetPhase()) {
        break
    }
}
```
</Tab>
<Tab value="Python">
```python
from ai.stigmer.agentic.workflowrun.v1.io_pb2 import SubscribeWorkflowRunRequest

for snapshot in client.workflow_runs.subscribe(
    SubscribeWorkflowRunRequest(run_id=workflow_run_id),
):
    for task in snapshot.status.tasks:
        print(f"{task.task_name}: {task.status}")

    if is_terminal(snapshot.status.phase):
        if snapshot.status.output:
            print("Output:", snapshot.status.output)
        break
```
</Tab>
<Tab value="Java">
```java
var stream = client.workflowRun.subscribe(
    SubscribeWorkflowRunRequest.newBuilder()
        .setRunId(workflowRunId)
        .build());

while (stream.hasNext()) {
    var snapshot = stream.next();
    var status = snapshot.getStatus();

    for (var task : status.getTasksList()) {
        System.out.printf("%s: %s%n", task.getTaskName(), task.getStatus());
    }

    if (isTerminal(status.getPhase())) {
        break;
    }
}
```
</Tab>
</SDKTabs>
